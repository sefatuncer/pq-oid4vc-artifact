  "∃ V I c #s #j.

    (¬(∃ #m. Migrate( I ) @ #m))"

  (MigratingIssuer( I ) @ #s) ∧ (AcceptClassical( V, I, c ) @ #j)

lemma executable_lifecycle:
  exists-trace
  "∃ V I c #m #s #j.
    (((Migrate( I ) @ #m) ∧ (Sunset( I ) @ #s)) ∧
     (AcceptPQ( V, I, c ) @ #j)) ∧
    (#s < #j)"
/*
guarded formula characterizing all satisfying traces:
"∃ V I c #m #s #j.
  (Migrate( I ) @ #m) ∧ (Sunset( I ) @ #s) ∧ (AcceptPQ( V, I, c ) @ #j)
 ∧
  #s < #j"
*/
simplify
solve( !IssPQ( I, kq ) ▶₀ #s )
  case Issuer_Setup_Born
  solve( Migrate( $I ) @ #m.2 )
    case Issuer_Setup_Born
    solve( AcceptPQ( V, $I, c ) @ #j )
      case Acc1_PQ_NoPolicy
      solve( !Anchor( A, pk(x), 'cl' ) ▶₀ #j )
        case ClassicalRoot_Setup
        solve( Migrate( $I ) @ #m )
          case Issuer_Setup_Born
          solve( Migrate( $I ) @ #m.1 )
            case Issuer_Setup_Born
            solve( !KU( sign(<'iss_cert', $I, pk(x), 'pq'>, ~kr) ) @ #vk.11 )
              case c_sign
              solve( !KU( ~kr ) @ #vk.17 )
