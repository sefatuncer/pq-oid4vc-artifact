lemma attack_needs_crqc:
  all-traces
  "∀ V I c #m #j.
    (((Migrate( I ) @ #m) ∧ (Accept( V, I, c ) @ #j)) ∧ (#m < #j)) ⇒
    ((∃ #i. (Issued( I, c ) @ #i) ∧ (#i < #j)) ∨
     (∃ k #b. (Broken( k ) @ #b) ∧ (#b < #j)))"
/*
guarded formula characterizing all counter-examples:
"∃ V I c #m #j.
  (Migrate( I ) @ #m) ∧ (Accept( V, I, c ) @ #j)
 ∧
  (#m < #j) ∧
  (∀ #i. (Issued( I, c ) @ #i) ⇒ ¬(#i < #j)) ∧
  (∀ k #b. (Broken( k ) @ #b) ⇒ ¬(#b < #j))"
*/
simplify
solve( Migrate( I ) @ #m )
  case Issuer_Setup_Born
  solve( !CASk( 'CAM', kcam ) ▶₀ #m )
    case CAM_Setup
    solve( Accept( V, $I, c ) @ #j )
      case Acc1_Classical
      solve( Exp( V, sid, $I, 'none' ) ▶₀ #j )
        case Exp_Fresh_None
        solve( !Anchor( A, pk(x), aA ) ▶₁ #j )
          case CAM_Setup
          solve( !KU( sign(<'iss_cert', $I, pk(x), 'cl'>, ~kcam) ) @ #vk.13 )
            case Issuer_Setup_Born
            solve( !KU( sign(<'cred', $I, c>, ~kc) ) @ #vk.17 )
              case Issue_Classical
              by contradiction /* from formulas */
            next
              case c_sign
              solve( !KU( ~kc ) @ #vk.23 )
                case CRQC_Break_Issuer_Classical
                by contradiction /* from formulas */
              qed
            qed
          next
            case c_sign
            by solve( !KU( ~kcam ) @ #vk.23 )
          qed
        next
          case OldCAM_Setup
          solve( !KU( sign(<'iss_cert', $I, pk(x), 'cl'>, ~ko) ) @ #vk.13 )
            case c_sign
            solve( !KU( ~ko ) @ #vk.23 )
              case CRQC_Break_Classical_Anchor
              by contradiction /* from formulas */
            qed
          qed
        qed
      next
        case Exp_Fresh_Req
        by contradiction /* from formulas */
      qed
    next
      case Acc1_PQ_Exp
      solve( Exp( V, sid, $I, e ) ▶₀ #j )
        case Exp_Fresh_None
        by contradiction /* from formulas */
      next
        case Exp_Fresh_Req
        solve( !Anchor( A, pk(x), aA ) ▶₁ #j )
          case CAM_Setup
          solve( !PinPQ( $I, pk(x) ) ▶₂ #j )
            case Issuer_Setup_Born
            solve( !KU( sign(<'cred', $I, c>, ~kq) ) @ #vk.17 )
              case Issue_PQ
              by contradiction /* from formulas */
            next
              case c_sign
              by solve( !KU( ~kq ) @ #vk.23 )
            qed
          qed
        next
          case OldCAM_Setup
          solve( !PinPQ( $I, pk(x) ) ▶₂ #j )
            case Issuer_Setup_Born
            solve( !KU( sign(<'iss_cert', $I, pk(~kq), 'pq'>, ~ko) ) @ #vk.14 )
              case c_sign
              solve( !KU( ~ko ) @ #vk.23 )
                case CRQC_Break_Classical_Anchor
                by contradiction /* from formulas */
              qed
            qed
          qed
        qed
